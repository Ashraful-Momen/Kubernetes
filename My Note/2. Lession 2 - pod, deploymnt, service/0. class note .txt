===================================== Deployment  =====================================

#deployment.yml: 
-----------------
apiVersion: apps/v1                  # API version for Deployment resources. 
                                     # 'apps/v1' is the stable API for Deployment, ReplicaSet, StatefulSet.
kind: Deployment                     # Declares the resource type = Deployment. 
                                     # Deployment manages ReplicaSets and ensures desired number of Pods.

metadata:                            # Metadata = basic information about this Deployment.
  name: nginx-deployment             # Name of the Deployment (must be unique in the namespace).
  labels:                            # Labels = key-value pairs to identify the Deployment.
    app: nginx                       # Label used to group and select this Deployment.

spec:                                # Specification = what this Deployment should create or manage.
  replicas: 3                        # Number of Pod replicas required. Deployment ensures 3 Pods always running.

  selector:                          # Tells Deployment WHICH Pods it should manage. select from other resource file where app = nginx; as like inside => create_pod.yml file use lavel 'app:nginx' and selector use this label from creat_pod.yml
    matchLabels:                     # Only Pods with these labels will be controlled.
      app: nginx                     # Must match template.metadata.labels exactly.

  template:                          # Pod template = blueprint for creating Pods.
                                     # Deployment uses this to create the Pods under it.
    metadata:
      labels:
        app: nginx                   # Labels added to Pods created by this Deployment.
                                     # Must match 'selector.matchLabels'.

    spec:                            # Pod specification (inside Deployment).
      containers:                    # List of containers inside each Pod.
        - name: nginx-container      # Name of the container inside the Pod.
          image: nginx:latest        # Container image from Docker Hub/local. 
                                     # 'latest' tag pulls the newest nginx image.

          ports:
            - containerPort: 80      # Container will listen on port 80.
                                     # Used for exposing service traffic inside the Pod.

====================================== Service : port-forward from deployment file =======================================

apiVersion: v1                   # API version for Service resources
kind: Service                    # declares this resource as a Service
metadata:
  name: nginx-service            # name of the Service
  namespace: ostad               # target namespace where this Service will be created
  labels:
    app: nginx                   # label to identify this Service (used for selectors, management, etc.)

spec:
  type: LoadBalancer             # Service type: LoadBalancer exposes service externally
                                 # On cloud: gets an external IP automatically
                                 # On minikube: requires `minikube tunnel` to get EXTERNAL-IP
  selector:                      # connects this Service to Pods with matching labels
    app: nginx                   # must match the `labels` in Deployment/Pod template
  ports:
    - protocol: TCP              # protocol used by this Service (TCP/UDP)
      port: 8080                 # Service port inside cluster (cluster-level access)
      targetPort: 80             # Pod/container port to which the traffic is forwarded


# Notes on Service types:
# type: LoadBalancer   --> publicly expose service (requires cloud LB or minikube tunnel)
# type: NodePort       --> accessible via <NodeIP>:<nodePort>, works locally
# type: ClusterIP      --> default, internal only, accessible within cluster
