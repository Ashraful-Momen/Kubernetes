
How to install helm: 
---------------------
# Install Helm on Parrot OS (Debian-based)

1. Add Helm GPG key:
curl https://baltocdn.com/helm/signing.asc | sudo apt-key add -

2. Add Helm repository:
echo "deb https://baltocdn.com/helm/stable/debian/ all main" | sudo tee /etc/apt/sources.list.d/helm-stable-debian.list

3. Update packages:
sudo apt update

4. Install Helm:
sudo apt install helm -y

5. Verify:
helm version

8.1. ingress.yaml (read for understanding): 
----------------
{{- if .Values.ingress.enabled }}                 # If ingress is enabled in values.yaml → render everything below

apiVersion: networking.k8s.io/v1                  # Kubernetes API version for Ingress
kind: Ingress                                      # Kubernetes resource kind = Ingress

metadata:
  name: {{ include "django-app.name" . }}-ingress # Ingress name → generated using helper function from _helpers.tpl

spec:
  rules:                                          # Starts the routing rules section
    - host:                                       # Defines host/domain name for incoming requests
        {{- if .Values.ingress.host }}            # If a custom hostname is set in values.yaml
            {{ .Values.ingress.host }}            # Use that hostname value
        {{ else }}                                # Otherwise (no custom host)
            "{{ include "django-app.name" . }}.example.com"   # Use default host based on chart name
        {{ end }}
      http:                                       # Defines HTTP routing
        paths:                                    # List of paths for this host
          - path: /                               # Match all requests starting with "/"
            pathType: Prefix                      # Prefix means /api also matches /, /dashboard etc.
            backend:                              # Backend service that receives the routed traffic
              service:                            # Use Kubernetes Service as backend
                name: {{ include "django-app.name" . }}-service # Service name → same base name as chart
                port:
                  number: {{ .Values.service.port }} # Port number from values.yaml → service configuration

{{- end }}                                         # End of the ingress.enabled condition — stops rendering if disabled

8.2. ingress.yaml (read for understanding): 
-------------------------------------------
{{- if .Values.ingress.enabled }}           
# IF STATEMENT:
# This checks: values.yaml → ingress.enabled
# If true → render this entire Ingress YAML
# If false → Helm outputs NOTHING for this file

apiVersion: networking.k8s.io/v1             
# Kubernetes API version for Ingress resources

kind: Ingress                                 
# We are creating an Ingress object

metadata:
  name: {{ include "django-app.name" . }}-ingress      
  # Creates Ingress name by calling helper:
  # include "django-app.name" → gets Chart.Name (from chart.yaml)
  # Example final result: myapp-ingress

spec:
  rules:                                      
  # Specifies the routing rules for the Ingress controller

    - host: {{- if .Values.ingress.host }}{{ .Values.ingress.host }}{{ else }}"{{ include "django-app.name" . }}.example.com"{{ end }}
      # This line decides the hostname using IF/ELSE:
      # If values.yaml has ingress.host set → USE THAT value
      # Else → fallback to "<chartName>.example.com"
      #
      # Example:
      # values.ingress.host = "api.my-domain.com"  → uses that
      # values.ingress.host = "" → uses "myapp.example.com"

      http:                                     
      # Indicates this rule applies to HTTP routing

        paths:                                  
        # List of path rules inside this host

          - path: /                              
            # Incoming requests matching "/" prefix come here
            # Covers "/", "/login", "/anything"

            pathType: Prefix                     
            # Prefix means: match ANY subpath starting with "/"

            backend:                             
            # Define where the request should be routed

              service:                           
              # Traffic goes to a Kubernetes Service

                name: {{ include "django-app.name" . }}-service   
                # Service name (must exist in cluster)
                # Example: myapp-service

                port:
                  number: {{ .Values.service.port }}              
                  # The service port is taken from values.yaml
                  # Example:
                  # service.port: 80 → final YAML: number: 80

{{- end }}
# END IF:
# Ends the "if .Values.ingress.enabled" condition started at the top.
# If ingress.enabled = false → everything above is NOT rendered.
